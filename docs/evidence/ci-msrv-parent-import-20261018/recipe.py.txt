H = lambda b: hashlib.sha256(b).hexdigest()
before = {}
def save(path):
    b = (R / path).read_bytes()
    q = O / 'before' / path
    assert not q.exists(), path
    q.parent.mkdir(parents=True, exist_ok=True)
    q.write_bytes(b)
    before[path] = {'length': len(b), 'sha256': H(b)}
    return b
def archive(raw):
    path = E + '/whole/' + H(raw) + '.txt'
    p = R / path
    p.parent.mkdir(parents=True, exist_ok=True)
    if p.exists(): assert p.read_bytes() == raw
    else: p.write_bytes(raw)
    return {'length': len(raw), 'sha256': H(raw), 'archive': path}
