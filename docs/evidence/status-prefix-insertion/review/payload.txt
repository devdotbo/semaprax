    fn probe(
        &self,
        function: &ResolvedFunction,
        key: &K,
        budget: &mut ReplayBudget,
    ) -> Result<Probe, Diagnostic> {
        let mut probe = Probe {
            found: false,
            parent: NIL,
            before: NIL,
            after: NIL,
            side: Ordering::Equal,
        };
        let mut at = self.root;
        while at != NIL {
            let node = &self.nodes[at];
            let units = key
                .comparison_units(&node.key)
                .and_then(|units| units.checked_add(VISIT_UNITS))
                .ok_or_else(|| {
                    overflow(function, budget, "directory key comparison work overflow")
                })?;
            budget.charge(function, units, self.phase)?;
            probe.side = Ord::cmp(key, &node.key);
            if probe.side == Ordering::Equal {
                probe.found = true;
                return Ok(probe);
            }
            probe.parent = at;
            if probe.side == Ordering::Less {
                probe.after = at;
                at = node.left;
            } else {
                probe.before = at;
                at = node.right;
            }
        }
        Ok(probe)
    }

    pub(super) fn contains(
        &self,
        function: &ResolvedFunction,
        key: &K,
        budget: &mut ReplayBudget,
    ) -> Result<bool, Diagnostic> {
        self.authenticate(function, budget)?;
        Ok(self.probe(function, key, budget)?.found)
    }

    /// Returns `false` for a duplicate, like `BTreeSet::insert`.
    pub(super) fn insert(
        &mut self,
        function: &ResolvedFunction,
        key: K,
        budget: &mut ReplayBudget,
    ) -> Result<bool, Diagnostic> {
        self.authenticate(function, budget)?;
        let probe = self.probe(function, &key, budget)?;
        if probe.found {
            return Ok(false);
        }
        self.grow(function, budget)?;
        budget.charge(function, VISIT_UNITS, self.phase)?;
        let index = self.nodes.len();
        self.nodes.push(Node {
            key,
            left: NIL,
            right: NIL,
            parent: probe.parent,
            next: probe.after,
            height: 1,
        });
        if probe.before == NIL {
            self.first = index;
        } else {
            self.nodes[probe.before].next = index;
        }
        if probe.parent == NIL {
            self.root = index;
        } else if probe.side == Ordering::Less {
            self.nodes[probe.parent].left = index;
        } else {
            self.nodes[probe.parent].right = index;
        }
        self.rebalance(probe.parent, function, budget)?;
        Ok(true)
    }

fn compare(
    function: &ResolvedFunction,
    query: &ExpressionId,
    candidate: &ExpressionId,
    prefix: usize,
    budget: &mut ReplayBudget,
) -> Result<(Ordering, usize), Diagnostic> {
    budget.reserve_bytes(function, COMPARISON_BYTES, "expression-fact comparison controls")?;
    budget.charge(function, BOUND_UNITS, "expression-fact prefix control bookkeeping")?;
    let mut comparison = Comparison {
        left: query.as_str().as_bytes(), right: candidate.as_str().as_bytes(),
        end: query.as_str().len().min(candidate.as_str().len()), common: prefix,
    };
    let units = comparison.end.checked_sub(prefix).and_then(|n| n.checked_add(1))
        .ok_or_else(|| {
            budget.budget_refusal(function, "expression-fact prefix comparison work overflow", "")
        })?;
    // Same one-unit-per-byte-pair comparison rate as the complete old lookup,
    // including its end comparison. Only certified bytes are no longer read.
    budget.charge(function, units, "expression-fact certified suffix lookup")?;
    while comparison.common < comparison.end {
        let a = comparison.left[comparison.common];
        let b = comparison.right[comparison.common];
        if a != b { return Ok((a.cmp(&b), comparison.common)); }
        comparison.common += 1;
    }
    Ok((comparison.left.len().cmp(&comparison.right.len()), comparison.common))
}
