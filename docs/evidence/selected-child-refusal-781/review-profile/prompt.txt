/// Only calls on ReplayBudget may construct this already-paid budget arm.
/// Finite semantic producers carry Copy data, never a Diagnostic conversion.
pub(in super::super) enum Deferred {
    Budget(crate::diagnostic::Diagnostic),
    Finite(Refusal),
}
impl From<Refusal> for Deferred {
    fn from(reason: Refusal) -> Self {
        Self::Finite(reason)
    }
}
impl<'a> Permit<'a> {
    pub(in super::super) fn profile<T>(
        subject: &'a ResolvedFunction,
        budget: &mut ReplayBudget,
    ) -> Result<Self, Error> {
        type Controls<T> = (
            Deferred,
            Option<Deferred>,
            Result<T, Deferred>,
            Result<T, Deferred>,
        );
        budget.reserve_array::<Controls<T>>(subject, 1, "selected profile child controls")?;
        budget.charge(subject, 4, "selected profile child controls")?;
        Self::acquire::<T>(subject, budget)
    }
    pub(in super::super) fn resolve<T>(self, result: Result<T, Deferred>) -> Result<T, Error> {
        match result {
            Ok(value) => Ok(value),
            Err(Deferred::Finite(reason)) => Err(self.refuse(reason)),
            Err(Deferred::Budget(error)) => Err(Error::Owned(error)),
        }
    }
}

pub(in crate::cleanup_plan::replay) fn append(
    function: &ResolvedFunction,
    binding: &crate::hir::ResolvedBinding,
    value: &ResolvedExpr,
    budget: &mut ReplayBudget,
) -> Result<bool, Diagnostic> {
    if budget.refusal_spent {
        return Err(budget.budget_refusal(function, "selected replay budget already failed", ""));
    }
    // Debit the allowance `ReplayBudget::charge` would use. A factored
    // comparison lends the whole free allowance to its skeleton constructors,
    // so `remaining` is then zero and only the lent allowance may pay.
    let ReplayBudget {
        remaining,
        skeleton_remaining,
        skeleton_spent,
        lent_skeleton,
        physical,
        ..
    } = budget;
    let lent = *lent_skeleton;
    let physical = physical
        .as_deref_mut()
        .ok_or_else(|| replay_error(function, "selected append lost reservation"))?;
    let mut exhausted = false;
    let answer = crate::cleanup_plan::renewal_profile::selected_string_append(
        binding,
        value,
        physical,
        &mut |units| {
            let debited = if lent {
                skeleton_remaining
                    .checked_sub(units)
                    .zip(skeleton_spent.checked_add(units))
                    .map(|(next, spent)| {
                        *skeleton_remaining = next;
                        *skeleton_spent = spent;
                    })
            } else {
                remaining.checked_sub(units).map(|next| *remaining = next)
            };
            debited.ok_or_else(|| {
                exhausted = true;
                // This marker is consumed below, never published. The outer
                // refusal owns the sole prepaid diagnostic String backing.
                Diagnostic::io("SPX-H006", String::new())
            })
        },
    );
    if exhausted {
        // Same sticky refusal as every other selected debit: no later member
        // or retry can spend what this exhausted allowance did not have.
        let error = budget.budget_refusal(
            function,
            "cleanup replay work budget exhausted during ",
            "selected String append",
        );
        exhaust(budget);
        return Err(error);
    }
    answer
}

    program: &dyn crate::cleanup::ReplayContext,
    ty: &'a ResolvedType,
) -> Result<Option<&'a ResolvedType>, Refusal> {
    let ResolvedType::Nominal {
        declaration,
        arguments,
    } = ty
    else {
        return Ok(None);
    };
    if declaration.as_str() != crate::prelude::VEC_ID {
        return Ok(None);
    }
    authenticate_selected_vector_inline(program, declaration)?;
    let [element] = arguments.as_slice() else {
        return Err(refusal(
            ProfileRefusal::SelectedVecHasInvalidTypeArgumentCount,
        ));
    };
    if crate::vec_ops::resolved_vec_element_is_admitted(element) || *element == ResolvedType::String
    {
        return Ok(Some(element));
    }
    if matches!(element, ResolvedType::Nominal { arguments, .. } if arguments.is_empty()) {
        selected_flat_record_fields_inline(program, element)?;
        if crate::hir::copy_record_collection::admitted(program.declarations(), element)
            || crate::hir::owned_leaf_collection::runtime_element(program.declarations(), element)
        {
            return Ok(Some(element));
        }
    }
    Err(refusal(
        ProfileRefusal::SelectedVecElementIsOutsideTheCurrentBoundedFlatFamily,
    ))
}
