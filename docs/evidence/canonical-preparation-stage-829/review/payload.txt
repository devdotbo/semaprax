pub(crate) fn with_limit<T>(limit: usize, operation: impl FnOnce() -> T) -> (T, bool) {
    let (value, overflowed, _) = with_limit_usage(limit, operation);
    (value, overflowed)
}

pub(crate) fn with_limit_usage<T>(limit: usize, operation: impl FnOnce() -> T) -> (T, bool, usize) {
    let (value, overflowed, consumed, _) = with_limit_usage_refusal(limit, operation);
    (value, overflowed, consumed)
}

pub(crate) fn with_limit_usage_refusal<T>(
    limit: usize,
    operation: impl FnOnce() -> T,
) -> (T, bool, usize, Option<ReservationRefusal>) {
    struct Restore {
        previous: Option<Rc<Budget>>,
        current: Rc<Budget>,
        initial: usize,
    }
    impl Drop for Restore {
        fn drop(&mut self) {
            let consumed = self.initial.saturating_sub(self.current.remaining.get());
            let previous = self.previous.take();
            ACTIVE.with(|active| {
                active.replace(previous.clone());
            });
            if let Some(parent) = previous {
                if !reserve(Some(&parent), consumed) {
                    record_refusal(&parent, consumed);
                    // A refused reservation normally means nothing was
                    // written, but here the child already emitted the bytes:
                    // they cannot be un-spent. Charge the parent anyway, so
                    // `consumed` and `active_remaining` stay truthful, and
                    // report the refusal.
                    let remaining = parent.remaining.get();
                    parent.remaining.set(remaining.saturating_sub(consumed));
                    #[cfg(test)]
                    stage_debits::debit(&parent, remaining.min(consumed));
                    parent.overflowed.set(true);
                }
            }
        }
    }
    let parent = active();
    let effective_limit = parent
        .as_ref()
        .map_or(limit, |budget| limit.min(budget.remaining.get()));
    let budget = Rc::new(Budget {
        initial: effective_limit,
        remaining: Cell::new(effective_limit),
        floor: Cell::new(0),
        overflowed: Cell::new(false),
        first_refusal: Cell::new(None),
        stage: Cell::new(None),
    });
    let previous = ACTIVE.with(|active| active.replace(Some(Rc::clone(&budget))));
    let restore = Restore {
        previous,
        current: Rc::clone(&budget),
        initial: effective_limit,
    };
    let value = operation();
    let overflowed = budget.overflowed.get();
    let consumed = effective_limit.saturating_sub(budget.remaining.get());
    let refusal = budget.first_refusal.get();
    drop(restore);
    (value, overflowed, consumed, refusal)
}

fn active() -> Option<Rc<Budget>> {
    ACTIVE.with(|active| active.borrow().clone())
}

fn record_refusal(budget: &Budget, length: usize) {
    if budget.first_refusal.get().is_none() {
        #[cfg(test)]
        stage_debits::refusal(budget, length);
        budget.first_refusal.set(Some(ReservationRefusal {
            requested: length,
            remaining: budget.remaining.get(),
            floor: budget.floor.get(),
            stage: budget.stage.get(),
        }));
    }
}

/// A stage belongs to the current budget only. Nested budgets start unlabelled,
/// and unwind restores the previous stage before the next reservation.
pub(crate) fn with_reservation_stage<T>(
    stage: ReservationStage,
    operation: impl FnOnce() -> T,
) -> T {
    let restore = reservation_stage_guard(stage);
    let value = operation();
    drop(restore);
    value
}

/// Fixed-size restoration handle. Cloning the existing budget Rc allocates no
/// backing, and the handle never changes budget ownership or refusal selection.
pub(crate) struct ReservationStageGuard(Option<(Rc<Budget>, Option<ReservationStage>)>);
impl Drop for ReservationStageGuard {
    fn drop(&mut self) {
        if let Some((budget, previous)) = self.0.take() {
            budget.stage.set(previous);
        }
    }
}
pub(crate) fn reservation_stage_guard(stage: ReservationStage) -> ReservationStageGuard {
    ReservationStageGuard(active().map(|budget| {
        let previous = budget.stage.replace(Some(stage));
        (budget, previous)
    }))

//! One actual caller ledger and its dedicated allocation-refusal text.
//! Acquisition creates no allowance; this affine owner never proves admission.
use super::*;
use crate::diagnostic::Diagnostic;
use std::alloc::Layout;
use std::mem::size_of;

const MESSAGE: &str = "canonical loan CFG storage reservation refused";

#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub(crate) enum Refused {
    NoOwner,
    Budget(ReservationRefusal),
    Allocation,
    Capacity,
    Detached,
    Arithmetic,
}

pub(crate) struct CfgReservation {
    budget: Rc<Budget>,
    text: String,
    first: Cell<Option<Refused>>,
}

impl CfgReservation {
    pub(crate) fn acquire() -> Result<Self, Refused> {
        let budget = active().ok_or(Refused::NoOwner)?;
        let bytes = size_of::<Self>()
            .checked_add(MESSAGE.len())
            .ok_or_else(|| {
                poison(&budget, usize::MAX);
                Refused::Arithmetic
            })?;
        required(&budget, bytes)?;
        let mut text = String::new();
        if text.try_reserve_exact(MESSAGE.len()).is_err() {
            poison(&budget, 0);
            return Err(Refused::Allocation);
        }
        if text.capacity() != MESSAGE.len() {
            required(&budget, text.capacity().saturating_sub(MESSAGE.len()))?;
            poison(&budget, 0);
            return Err(Refused::Capacity);
        }
        Ok(Self {
            budget,
            text,
            first: Cell::new(None),
        })
    }

    pub(crate) fn reserve(&self, bytes: usize) -> Result<(), Refused> {
        // Identity is checked even for zero bytes and spare-capacity requests.
        if !active().is_some_and(|current| Rc::ptr_eq(&current, &self.budget)) {
            return self.fail(Refused::Detached, bytes);
        }
        if let Some(first) = self.first.get() {
            return Err(first);
        }
        required(&self.budget, bytes).inspect_err(|&reason| {
            self.first.set(Some(reason));
        })
    }

    pub(crate) fn array<T>(&self, count: usize) -> Result<(), Refused> {
        self.reserve(0)?;
        match Layout::array::<T>(count) {
            Ok(layout) => self.reserve(layout.size()),
            Err(_) => self.fail(Refused::Arithmetic, usize::MAX),
        }
    }

    pub(crate) fn capacity(&self, requested: usize, excess: usize) -> Result<(), Refused> {
        self.reserve(excess)?;
        self.fail(Refused::Capacity, requested)
    }

    pub(crate) fn allocation(&self, requested: usize) -> Refused {
        self.fail(Refused::Allocation, requested).unwrap_err()
    }

    pub(crate) fn arithmetic(&self) -> Refused {
        self.fail(Refused::Arithmetic, usize::MAX).unwrap_err()
    }

    pub(crate) fn first(&self) -> Option<Refused> {
        self.first.get()
    }

    fn fail(&self, reason: Refused, requested: usize) -> Result<(), Refused> {
        let first = self.first.get().unwrap_or(reason);
        self.first.set(Some(first));
        poison(&self.budget, requested);
        Err(first)
    }

    /// Consumes this exact owner; no new String or formatting after refusal.
    pub(crate) fn diagnostic(mut self) -> Diagnostic {
        debug_assert!(self.first.get().is_some());
        self.text.push_str(MESSAGE);
        Diagnostic::io("SPX-H006", self.text)
    }

    #[cfg(test)]
    pub(crate) fn text_pointer(&self) -> *const u8 {
        self.text.as_ptr()
    }
}

fn poison(budget: &Budget, requested: usize) {
    record_refusal(budget, requested);
    budget.overflowed.set(true);
}
fn required(budget: &Budget, bytes: usize) -> Result<(), Refused> {
    if reserve(Some(budget), bytes) {
        return Ok(());
    }
    poison(budget, bytes);
    Err(Refused::Budget(
        budget
            .first_refusal
            .get()
            .expect("required refusal recorded"),
    ))
}

#[derive(Clone, Copy)]
enum Payer<'payer> {
    Captured(&'payer CfgReservation),
    ActiveBuilder,
    LegacyUnobserved,
}
impl Payer<'_> {
    fn current() -> Self {
        if crate::bounded_output::active_remaining().is_some() {
            Self::ActiveBuilder
        } else {
            Self::LegacyUnobserved
        }
    }
    fn reserve(self, bytes: usize) -> Result<(), Diagnostic> {
        if let Self::Captured(owner) = self {
            return owner.reserve(bytes).map_err(|_| refused());
        }
        if matches!(self, Self::ActiveBuilder)
            && !crate::bounded_output::reserve_active_required(bytes)
        {
            return Err(refused());
        }
        Ok(())
    }
    fn capacity(self, requested: usize, excess: usize) -> Diagnostic {
        if let Self::Captured(owner) = self {
            let _ = owner.capacity(requested, excess);
            return refused();
        }
        if matches!(self, Self::ActiveBuilder) {
            if excess != 0 {
                crate::bounded_output::reserve_active_required(excess);
            }
            crate::bounded_output::refuse_active_required(requested);
        }
        refused()
    }
}
fn refused() -> Diagnostic {
    Diagnostic::io("SPX-H006", String::new())
}

fn backing<T>(items: usize, payer: Payer<'_>) -> Result<usize, Diagnostic> {
    std::alloc::Layout::array::<T>(items)
        .map(|layout| layout.size())
        .map_err(|_| payer.capacity(usize::MAX, 0))
}
fn ensure<T>(values: &mut Vec<T>, needed: usize, payer: Payer<'_>) -> Result<(), Diagnostic> {
    // Even spare-capacity and zero requests preserve the selected first refusal.
    payer.reserve(0)?;
    if needed <= values.capacity() {
        return Ok(());
    }
    let capacity = needed.max(
        values
            .capacity()
            .checked_mul(2)
            .ok_or_else(|| payer.capacity(usize::MAX, 0))?,
    );
    let bytes = backing::<T>(capacity, payer)?;
    payer.reserve(bytes)?;
    if values.try_reserve_exact(capacity - values.len()).is_err() {
        return Err(payer.capacity(bytes, 0));
    }
    if values.capacity() != capacity {
        let excess =
            backing::<T>(values.capacity().saturating_sub(capacity), payer).unwrap_or(usize::MAX);
        return Err(payer.capacity(bytes, excess));
    }
    Ok(())
}
fn ensure_probe<T>(values: &mut Vec<T>, needed: usize, payer: Payer<'_>) -> Result<(), Diagnostic> {
    // Even spare-capacity and zero requests preserve the selected first refusal.
    payer.reserve(0)?;
    if needed <= values.capacity() {
        return Ok(());
    }
    // Probe entries are drained every append. Exact replacement avoids
    // doubling beyond the next complete plan while retaining prepaid backing.
    let capacity = needed;
    let bytes = backing::<T>(capacity, payer)?;
    payer.reserve(bytes)?;
    if values.try_reserve_exact(capacity - values.len()).is_err() {
        return Err(payer.capacity(bytes, 0));
    }
    if values.capacity() != capacity {
        let excess =
            backing::<T>(values.capacity().saturating_sub(capacity), payer).unwrap_or(usize::MAX);
        return Err(payer.capacity(bytes, excess));
    }
    Ok(())
}
fn count(current: usize, added: usize, payer: Payer<'_>) -> Result<usize, Diagnostic> {
    current
        .checked_add(added)
        .ok_or_else(|| payer.capacity(usize::MAX, 0))
}

pub(in crate::hir) fn validate_with_comparison<'a>(

        program,
        fused,
        &helpers,
        validator,
        &observation,
    )?;

fn prepare_sealed<'a, T>(
    seed: &ConstructorSeed<'_>,
    program: &'a ResolvedProgram,
    seal: impl FnOnce(Storage<'a>, Payer<'_>) -> Result<T, Diagnostic>,
) -> Result<(T, Deferred<'a>), Diagnostic> {
    let _stage = crate::bounded_output::reservation_stage_guard(
        crate::bounded_output::ReservationStage::CanonicalLoanStorage,
    );
    #[cfg(test)]
    let _operation = crate::bounded_output::stage_debits::operation_guard(
        crate::bounded_output::stage_debits::Operation::CanonicalLoanStorage,
    );
    if !std::ptr::eq(seed.program(), program) {
        return Err(hir_error(
            "canonical shared-loan constructor belongs to another program",
        ));
    }
    // Acquisition follows the unchanged ModuleIndex and ConstructorSeed
    // phases. One report/ledger is held across ALL plans and their append; a
    // storage-only test injection keeps its separate legacy payer below.
    let captured = if crate::bounded_output::active_remaining().is_some() {
        Some(CfgReservation::acquire().map_err(|_| refused())?)
    } else {
        None
    };
    let payer = captured
        .as_ref()
        .map_or_else(Payer::current, Payer::Captured);
    let result = (|| {
        payer.reserve(0)?;
        let mut deferred = Deferred::new(program);
        let mut storage = Storage {
            program,
            ids: Vec::new(),
            liveness: Vec::new(),
            origins: Vec::new(),
            _scratch: Scratch::default(),
        };
        for function in program
            .functions
            .iter()
            .chain(program.function_instances.iter().map(|i| &i.function))
        {
            let plan = match captured.as_ref() {
                Some(owner) => {
                    match crate::loan_plan::build_plan_with_reservation(program, function, owner) {
                        Ok(plan) => plan,
                        Err(crate::loan_plan::FundedError::Physical(reason)) => {
                            // The tag is copied from this exact owner's first failure.
                            debug_assert_eq!(owner.first(), Some(reason));
                            return Err(refused());
                        }
                        Err(crate::loan_plan::FundedError::Semantic(error)) => return Err(error),
                    }
                }
                None => crate::loan_plan::build_plan(program, function)?,
            };
            // Compare the COMPLETE plan before append consumes it. An
            // attached plan is never a sizing/admission source.
            deferred.compare(function, &plan);
            append(&mut storage, plan, payer)?;
        }
        Ok((seal(storage, payer)?, deferred))
    })();
    // Only this exact affine owner can mark physical failure. Never infer
    // payment from H006 text, and never reserve/render after semantic error.
    if let Some(owner) = captured {
        if owner.first().is_some() {
            return Err(owner.diagnostic());
        }
    }
    result
}

fn append(storage: &mut Storage<'_>, plan: LoanPlan, payer: Payer<'_>) -> Result<(), Diagnostic> {
    payer.reserve(0)?;
    // Detaching only owned backing lets the original complete receiver read
    // persistent directories without holding a mutable borrow of their owner.
    let mut scratch = std::mem::take(&mut storage._scratch);
    let result = append_reusing(storage, plan, payer, &mut scratch);
    scratch.local.clear();
    scratch.liveness.clear();
    scratch.ids.clear();
    // Success and refusal retain the same backing through receiving Drop.
    storage._scratch = scratch;
    result
}

fn append_reusing(
    storage: &mut Storage<'_>,
    plan: LoanPlan,
    payer: Payer<'_>,
    scratch: &mut Scratch,
) -> Result<(), Diagnostic> {
    let base = storage.origins.len();
    scratch.local.clear();
    let local = &mut scratch.local;
    ensure_probe(local, plan.loans.len(), payer)?;
    for (offset, loan) in plan.loans.iter().enumerate() {
        local.push((
            loan.id,
            base.checked_add(offset)
                .ok_or_else(|| hir_error("canonical shared-loan origin directory overflows"))?,
        ));
    }
    local.sort_unstable();
    let origin = |id: &LoanId| {
        // Complete original collect semantics: the final local occurrence wins.
        let end = local.partition_point(|entry| entry.0 <= *id);
        end.checked_sub(1)
            .and_then(|i| local.get(i))
            .filter(|entry| entry.0 == *id)
            .map(|entry| entry.1)
    };
    let mut live_count = 0usize;
    for endpoint in &plan.endpoints {
        let live = match endpoint.point.phase {
            LoanPointPhase::Before => &endpoint.live_before,
            LoanPointPhase::After => &endpoint.live_after,
        };
        live_count = count(live_count, live.len(), payer)?;
    }
    scratch.liveness.clear();
    let live_probe = &mut scratch.liveness;
    ensure_probe(live_probe, live_count, payer)?;
    for endpoint in &plan.endpoints {
        let live = match endpoint.point.phase {
            LoanPointPhase::Before => &endpoint.live_before,
            LoanPointPhase::After => &endpoint.live_after,
        };
        for loan in live {
            let source = live_probe.len();
            live_probe.push((
                (
                    endpoint.point.expression.clone(),
                    endpoint.point.phase,
                    *loan,
                ),
                origin(loan),
                source,
            ));
        }
    }
    // This private ordering indexes first source occurrences, not canonical
    // endpoint/loan vectors or error selection. It requires no sort scratch.
    live_probe.sort_unstable_by(|a, b| a.0.cmp(&b.0).then(a.2.cmp(&b.2)));
    let mut source = 0usize;
    for endpoint in &plan.endpoints {
        let live = match endpoint.point.phase {
            LoanPointPhase::Before => &endpoint.live_before,
            LoanPointPhase::After => &endpoint.live_after,
        };
        for loan in live {
            origin(loan)
                .ok_or_else(|| hir_error("canonical shared-loan endpoint names an unknown loan"))?;
            let key = (
                endpoint.point.expression.clone(),
                endpoint.point.phase,
                *loan,
            );
            let first = live_probe.partition_point(|entry| entry.0 < key);
            if storage.live(&key).is_some() || live_probe[first].2 < source {
                return Err(hir_error("canonical shared-loan liveness is duplicated"));
            }
            source += 1;
        }
    }
    scratch.ids.clear();
    let id_probe = &mut scratch.ids;
    ensure_probe(id_probe, plan.loans.len(), payer)?;
    id_probe.extend(
        plan.loans
            .iter()
            .enumerate()
            .map(|(source, loan)| ((loan.site.clone(), loan.cause.clone()), loan.id, source)),
    );
    id_probe.sort_unstable_by(|a, b| a.0.cmp(&b.0).then(a.2.cmp(&b.2)));
    for (source, loan) in plan.loans.iter().enumerate() {
        let key = (loan.site.clone(), loan.cause.clone());
        let first = id_probe.partition_point(|entry| entry.0 < key);
        if loan.start.phase != LoanPointPhase::Before
            || storage.id(&key).is_some()
            || id_probe[first].2 < source
        {
            return Err(hir_error(
                "canonical shared-loan identity is ambiguous or malformed",
            ));
        }
    }
    // No partial plan directory is published. Growth prepays full replacement
    // capacities while old backing is still live, never a delta or refund.
    let id_needed = count(storage.ids.len(), id_probe.len(), payer)?;
    let live_needed = count(storage.liveness.len(), live_probe.len(), payer)?;
    let origin_needed = count(storage.origins.len(), plan.loans.len(), payer)?;
    ensure(&mut storage.ids, id_needed, payer)?;
    ensure(&mut storage.liveness, live_needed, payer)?;
    ensure(&mut storage.origins, origin_needed, payer)?;
    storage
        .ids
        .extend(id_probe.drain(..).map(|(key, id, _)| (key, id)));
    storage.liveness.extend(
        live_probe
            .drain(..)
            .map(|(key, ordinal, _)| (key, ordinal.expect("checked every source endpoint"))),
    );
    storage
        .origins
        .extend(plan.loans.into_iter().map(|loan| loan.origin));
    storage.ids.sort_unstable_by(|a, b| a.0.cmp(&b.0));
    storage.liveness.sort_unstable_by(|a, b| a.0.cmp(&b.0));
    Ok(())
}
