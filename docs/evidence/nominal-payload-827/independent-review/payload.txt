5:     root: &ResolvedType,
6:     reserve: &mut impl FnMut(usize) -> Result<(), E>,
7: ) -> Result<usize, E> {
8:     fn push<'a, E>(
9:         stack: &mut Vec<&'a ResolvedType>,
10:         ty: &'a ResolvedType,
11:         reserve: &mut impl FnMut(usize) -> Result<(), E>,
12:     ) -> Result<(), E> {
13:         if stack.len() == stack.capacity() {
14:             let capacity = stack.capacity().saturating_mul(2).max(4);
15:             reserve(
16:                 size_of::<Vec<&ResolvedType>>()
17:                     .saturating_add(capacity.saturating_mul(size_of::<&ResolvedType>())),
18:             )?;
19:             let mut next = Vec::with_capacity(capacity);
20:             assert_eq!(next.capacity(), capacity);
21:             next.extend(stack.drain(..));
22:             *stack = next;
23:         }
24:         stack.push(ty);
25:         Ok(())
26:     }
27:     // Leaves require neither a census heap nor last-owner Drop scratch.
28:     if !matches!(root, ResolvedType::Nominal { arguments, .. } if !arguments.is_empty())
29:         && !matches!(root, ResolvedType::Function { .. })
30:     {
31:         return Ok(0);
32:     }
33:     let mut stack = Vec::new();
34:     push(&mut stack, root, reserve)?;
35:     let mut highwater = 0;
36:     while let Some(ty) = stack.pop() {
37:         match ty {
38:             ResolvedType::Nominal { arguments, .. } => {
39:                 for child in arguments {
40:                     push(&mut stack, child, reserve)?;
41:                 }
42:             }

45:                     push(&mut stack, child, reserve)?;

50:         }
51:         highwater = highwater.max(stack.len());
52:     }

149:     pub(super) fn acquire<E>(
150:         program: &dyn ProgramSource,
151:         prelude: &[crate::ast::TypeDeclaration],
152:         reserve: &mut impl FnMut(usize) -> Result<(), E>,
153:     ) -> Result<Self, E> {
154:         // Replacing Vec map values enlarges actual BTree nodes. Acquire a
155:         // complete current node upper bound per possible insertion in each
156:         // changed map, in addition to the unchanged caller prebound. There is
157:         // no credit for the old map or for omitted payload clones.
158:         fn node<V>() -> usize {
159:             crate::cleanup_plan::replay_btree_node_upper::<DeclarationId, V>()
160:                 .expect("fixed nominal map layout")
161:         }
162:         for declaration in program.types().iter().chain(prelude) {
163:             reserve(node::<NominalPayload<ResolvedTypeParameterDeclaration>>())?;
164:             match &declaration.kind {
165:                 TypeDeclarationKind::Record { .. } | TypeDeclarationKind::Class { .. } => {
166:                     reserve(node::<NominalPayload<ResolvedFieldDeclaration>>())?;
167:                 }
168:                 TypeDeclarationKind::Variant { cases } => {
169:                     reserve(node::<NominalPayload<ResolvedVariantCaseDeclaration>>())?;
170:                     for _ in cases {
171:                         reserve(node::<CaseFields>())?;
172:                     }
173:                 }
174:                 TypeDeclarationKind::Resource { .. } => {}
175:             }
176:         }
177:         for _ in program.functions() {
178:             reserve(node::<NominalPayload<ResolvedTypeParameterDeclaration>>())?;
179:         }
180:         for declaration in program.types() {
181:             if let TypeDeclarationKind::Class { methods, .. } = &declaration.kind {
182:                 // The complete old producer rebuilds this class field entry.
183:                 reserve(node::<NominalPayload<ResolvedFieldDeclaration>>())?;
184:                 for _ in methods {
185:                     reserve(node::<NominalPayload<ResolvedTypeParameterDeclaration>>())?;
186:                 }
187:             }
188:         }
189:         // Source types resolve shape-for-shape (parameters become scalar
190:         // metadata leaves); inherited fields reuse those same type shapes.
191:         // Traverse every declared field, not a nominal/prelude name filter.
192:         let mut capacity = 0;
193:         for declaration in program.types().iter().chain(prelude) {
194:             match &declaration.kind {
195:                 TypeDeclarationKind::Record { fields }
196:                 | TypeDeclarationKind::Class { fields, .. } => {
197:                     for field in fields {
198:                         capacity = capacity.max(frontier(&field.ty, reserve)?);
199:                     }
200:                 }
201:                 TypeDeclarationKind::Variant { cases } => {
202:                     for case in cases {
203:                         for field in &case.fields {
204:                             capacity = capacity.max(frontier(&field.ty, reserve)?);
205:                         }
206:                     }
207:                 }
208:                 TypeDeclarationKind::Resource { .. } => {}
209:             }
210:         }
211:         #[repr(C)]
212:         struct ArcCarrier<T> {
213:             strong: usize,
214:             weak: usize,
215:             value: T,
216:         }
217:         // At most one outer cases guard, one fields guard and one resolved
218:         // field type coexist. Price both initialization and receiving slots.
219:         let guards = 2 * size_of::<PendingVec<ResolvedVariantCaseDeclaration>>()
220:             + 2 * size_of::<PendingVec<ResolvedFieldDeclaration>>()
221:             + 2 * size_of::<PendingType>();
222:         reserve(
223:             size_of::<ArcCarrier<Mutex<Vec<ResolvedType>>>>()
224:                 .saturating_add(guards)
225:                 .saturating_add(size_of::<Self>())
226:                 .saturating_add(size_of::<Result<Self, E>>())
227:                 .saturating_add(capacity.saturating_mul(size_of::<ResolvedType>())),
228:         )?;
229:         let scratch = Vec::with_capacity(capacity);
230:         assert_eq!(scratch.capacity(), capacity);
231:         Ok(Self {
232:             scratch: Some(Arc::new(Mutex::new(scratch))),
233:             building: true,
234:         })
235:     }
236: }
