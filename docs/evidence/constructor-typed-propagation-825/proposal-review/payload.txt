//! One actual core-attempt ledger owns the single-diagnostic receiving slot.
//! This certifies neither existing multi-error helpers nor final legacy output.
use super::Failure;
use crate::bounded_output::{ClassifierLedger, active_remaining, reserve_active_required};
use crate::diagnostic::Diagnostic;
use std::mem::size_of;

#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub(in crate::workspace_graph) enum Setup {
    NoOwner,
    Budget { maximum: usize },
    Allocation,
    Capacity,
    Detached,
    Arithmetic,
}

/// Affine physical backing; no Clone, factory budget, or source authority.
pub(in crate::workspace_graph) struct Receiver {
    ledger: ClassifierLedger,
    report: Vec<Diagnostic>,
    detached: bool,
}
impl std::fmt::Debug for Receiver {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        f.debug_struct("ConstructorReceiver")
            .field("len", &self.report.len())
            .finish_non_exhaustive()
    }
}
#[derive(Debug)]
#[allow(
    clippy::large_enum_variant,
    reason = "the complete primary and unreceived owner must remain inline after refusal"
)]
pub(in crate::workspace_graph) enum CoreFailure {
    /// Only this one slot has genuine constructor receiving ownership.
    Received {
        report: Vec<Diagnostic>,
        ledger: ClassifierLedgerDebug,
    },
    /// Existing full ordered Vec backing; explicitly not recertified paid.
    Legacy(Vec<Diagnostic>),
    Setup(Setup),
    Unreceived {
        primary: Failure,
        owner: Receiver,
        refusal: Setup,
    },
}
/// Diagnostic formatting must not expose ledger access or allocate a proxy.
pub(in crate::workspace_graph) struct ClassifierLedgerDebug(ClassifierLedger);
impl std::fmt::Debug for ClassifierLedgerDebug {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        let _held = &self.0;
        f.write_str("held original constructor ledger")
    }
}
impl From<Vec<Diagnostic>> for CoreFailure {
    fn from(report: Vec<Diagnostic>) -> Self {
        Self::Legacy(report)
    }
}
impl From<Setup> for CoreFailure {
    fn from(failure: Setup) -> Self {
        Self::Setup(failure)
    }
}
impl Receiver {
    pub(in crate::workspace_graph) fn setup_bytes() -> Option<usize> {
        [
            size_of::<Self>(),
            size_of::<Option<Self>>(),
            size_of::<Result<Self, Setup>>(),
            size_of::<CoreFailure>(),
            size_of::<Vec<Diagnostic>>(),
            size_of::<Result<super::super::ResolvedCore, CoreFailure>>(),
            size_of::<
                Result<
                    (
                        super::super::ResolvedCore,
                        super::super::custody::SourceCustody,
                    ),
                    CoreFailure,
                >,
            >(),
            size_of::<Diagnostic>(),
        ]
        .into_iter()
        .try_fold(0usize, usize::checked_add)
    }
    pub(in crate::workspace_graph) fn acquire() -> Result<Self, Setup> {
        if active_remaining().is_none() {
            return Err(Setup::NoOwner);
        }
        let bytes = Self::setup_bytes().ok_or(Setup::Arithmetic)?;
        if !reserve_active_required(bytes) {
            return Err(Setup::Budget {
                maximum: super::super::active_builder_limit(),
            });
        }
        let ledger = ClassifierLedger::capture();
        let mut report = Vec::new();
        report.try_reserve_exact(1).map_err(|_| Setup::Allocation)?;
        if report.capacity() != 1 {
            let excess = report
                .capacity()
                .saturating_sub(1)
                .checked_mul(size_of::<Diagnostic>())
                .ok_or(Setup::Arithmetic)?;
            if !reserve_active_required(excess) {
                return Err(Setup::Budget {
                    maximum: super::super::active_builder_limit(),
                });
            }
            return Err(Setup::Capacity);
        }
        Ok(Self {
            ledger,
            report,
            detached: false,
        })
    }
    /// A selected primary may move into existing backing even when the held
    /// ledger is sticky. This authenticates identity and preserves ledger state;
    /// performs NO new reservation/formatting/output acquisition after Err.
    pub(in crate::workspace_graph) fn receive(mut self, failure: Failure) -> CoreFailure {
        if self.detached || !self.ledger.current() {
            self.detached = true;
            return CoreFailure::Unreceived {
                primary: failure,
                owner: self,
                refusal: Setup::Detached,
            };
        }
        match failure {
            Failure::One(primary) => {
                self.report.push(primary);
                CoreFailure::Received {
                    report: self.report,
                    ledger: ClassifierLedgerDebug(self.ledger),
                }
            }
            Failure::Many(report) => CoreFailure::Legacy(report),
        }
    }
}

//! Complete field evaluation and paid lazy Vec growth, with inline errors.
use super::{active_builder_limit, limit_error, FieldInitializer, Span, TypeDeclaration};
use crate::ast::FieldDeclaration;
use crate::workspace_graph::constructor_failure::{reserve, Failure};
fn physical() -> Failure {
    crate::bounded_output::refuse_active_required(usize::MAX);
    Failure::one(limit_error("builder_bytes", active_builder_limit()))
}
fn make(
    field: &FieldDeclaration,
    declarations: &[(&str, &TypeDeclaration)],
    allow_owned: bool,
) -> Result<FieldInitializer, Failure> {
    let span = Span::default();
    reserve(std::mem::size_of::<Result<FieldInitializer, Failure>>())?;
    reserve(std::mem::size_of::<FieldInitializer>())?;
    Ok(FieldInitializer {
        name: crate::bounded_output::budgeted_clone(&field.name),
        name_span: span,
        value: super::inline::default_expr(&field.ty, declarations, allow_owned)?,
        span,
    })
}
fn append(output: &mut Vec<FieldInitializer>, field: FieldInitializer) -> Result<(), Failure> {
    reserve(std::mem::size_of::<Result<(), Failure>>())?;
    reserve(0)?;
    if output.len() == output.capacity() {
        // Same non-byte minimum4 and geometric growth as the old Result<Vec>
        // collection. Acquire only after this full field succeeds, before push.
        let capacity = output.capacity().checked_mul(2).ok_or_else(physical)?.max(
            if std::mem::size_of::<FieldInitializer>() <= 1024 {
                4
            } else {
                1
            },
        );
        let layout =
            std::alloc::Layout::array::<FieldInitializer>(capacity).map_err(|_| physical())?;
        reserve(layout.size())?;
        output
            .try_reserve_exact(capacity - output.len())
            .map_err(|_| physical())?;
        if output.capacity() != capacity {
            return Err(physical());
        }
    }
    output.push(field);
    Ok(())
}
pub(in crate::workspace_graph) fn build(
    fields: &[FieldDeclaration],
    declarations: &[(&str, &TypeDeclaration)],
    allow_owned: bool,
) -> Result<Vec<FieldInitializer>, Failure> {
    reserve(std::mem::size_of::<Result<Vec<FieldInitializer>, Failure>>())?;
    reserve(0)?;
    let Some((first, rest)) = fields.split_first() else {
        reserve(std::mem::size_of::<Vec<FieldInitializer>>())?;
        return Ok(Vec::new());
    };
    // The old fallible iterator builds the complete first item before creating
    // its Vec. Preserve that semantic/error frontier before the modeled carrier.
    let first = make(first, declarations, allow_owned)?;
    reserve(std::mem::size_of::<Vec<FieldInitializer>>())?;
    let mut output = Vec::new();
    append(&mut output, first)?;
    for field in rest {
        append(&mut output, make(field, declarations, allow_owned)?)?;
    }
    reserve(0)?;
    Ok(output)
}
