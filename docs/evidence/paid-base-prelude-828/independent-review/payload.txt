194:     fn next(&mut self) -> Option<Self::Item> {

196:             return None;
197:         }
198:         let base = self.base.first();
199:         let local = self.local.peek().copied();
200:         let result = match (base, local) {
201:             (None, None) => return None,
202:             (Some(base), Some(local)) => {
203:                 if let Err(error) = compare(base.id.as_str(), local.id.as_str(), self.payer) {
204:                     self.failed = true;
205:                     return Some(Err(error));
206:                 }
207:                 match base.id.cmp(&local.id) {
208:                     std::cmp::Ordering::Greater => self.local.next().unwrap(),
209:                     std::cmp::Ordering::Equal => {
210:                         self.local.next();
211:                         self.base = &self.base[1..];
212:                         base
213:                     }
214:                     std::cmp::Ordering::Less => {
215:                         self.base = &self.base[1..];
216:                         base
217:                     }
218:                 }
219:             }
220:             (Some(base), None) => {
221:                 if let Err(error) = self.payer.work(1) {
222:                     self.failed = true;
223:                     return Some(Err(Failure::Payer(error)));
224:                 }
225:                 self.base = &self.base[1..];
226:                 base
227:             }
228:             (None, Some(_)) => {

232:                 }

118:     payer.work(1).map_err(Failure::Payer)?;
119:     let mut declarations = paid::vector(count().ok_or(Failure::Layout)?, payer)?;
120:     for schema in BASE {
121:         let top = descriptor(
122:             schema.id,
123:             schema.name,
124:             DeclarationKind::Variant,
125:             None,
126:             payer,
127:         )?;
128:         let owner = top.id.clone();
129:         declarations.push(top);
130:         for case in schema.cases {
131:             let child = descriptor(
132:                 case.id,
133:                 case.name,
134:                 DeclarationKind::VariantCase,
135:                 Some(owner.clone()),
136:                 payer,
137:             )?;
138:             let case_owner = child.id.clone();
139:             declarations.push(child);
140:             for field in case.fields {
141:                 declarations.push(descriptor(
142:                     field.id,
143:                     field.name,
144:                     DeclarationKind::CaseField,
145:                     Some(case_owner.clone()),
146:                     payer,
147:                 )?);
148:             }
149:         }
150:     }
151:     // Fallible insertion sort performs no hidden allocation and charges each
152:     // actual key comparison before reading its bytes. IDs must remain unique.
153:     for i in 1..declarations.len() {
154:         let mut j = i;
155:         while j > 0 {
156:             payer
157:                 .work(
158:                     declarations[j - 1]
159:                         .id
160:                         .as_str()
161:                         .len()
162:                         .checked_add(declarations[j].id.as_str().len())
163:                         .ok_or(Failure::Layout)?,
164:                 )
165:                 .map_err(Failure::Payer)?;
166:             match declarations[j - 1].id.cmp(&declarations[j].id) {
167:                 std::cmp::Ordering::Equal => return Err(Failure::Schema),
168:                 std::cmp::Ordering::Less => break,
169:                 std::cmp::Ordering::Greater => declarations.swap(j - 1, j),
170:             }
171:             j -= 1;
172:         }
173:     }
174:     let mut types = paid::vector(BASE.len(), payer)?;
175:     for schema in BASE {
176:         types.push(nominal(schema, &declarations, payer)?);
177:     }
178:     Ok(Contents {
179:         declarations,
180:         types,
181:     })
182: }

61:     /// Vec backing/payload must already belong to the caller. Refusal leaves
62:     /// that complete Vec untouched; only successful acquisition moves it. This acquires
63:     /// census scratch, last-owner disposal scratch, Arc backing, initializer,
64:     /// and the receiving carrier before each corresponding allocation.
65:     pub fn from_funded_vec<E>(
66:         values: &mut Vec<T>,
67:         mut reserve: impl FnMut(usize) -> Result<(), E>,
68:     ) -> Result<Self, E> {
69:         let mut capacity = 0;
70:         for value in values.iter() {
71:             capacity = capacity.max(value.drop_capacity(&mut reserve)?);
72:         }
73:         #[repr(C)]
74:         struct ArcCarrier<T> {
75:             strong: usize,
76:             weak: usize,
77:             value: T,
78:         }
79:         let backing = Layout::new::<ArcCarrier<Owner<T>>>().size();
80:         let bytes = backing
81:             .saturating_add(size_of::<Owner<T>>())
82:             .saturating_add(size_of::<Self>())
83:             .saturating_add(size_of::<Result<Self, E>>())
84:             .saturating_add(capacity.saturating_mul(size_of::<ResolvedType>()));

87:         assert_eq!(
88:             scratch.capacity(),
89:             capacity,
90:             "nominal exact disposal backing"
91:         );

94:                 values: std::mem::take(values),

97:         })
98:     }
99:     /// Explicit paid copy before mutation. Refusal leaves both old owners and
100:     /// their complete values unchanged; no implicit Cow/Arc::make_mut exists.
101:     pub fn try_make_mut<E>(
102:         &mut self,
103:         mut reserve: impl FnMut(usize) -> Result<(), E>,
104:     ) -> Result<&mut Vec<T>, E> {
105:         if matches!(&self.storage, Storage::Shared(_)) {
106:             let bytes = self.len().checked_mul(size_of::<T>()).and_then(|initial| {
107:                 self.iter()
108:                     .try_fold(initial, |n, v| n.checked_add(v.clone_bytes()?))

113:                 values.capacity(),
114:                 self.len(),
115:                 "nominal exact materialization backing"
116:             );

119:         } else {
120:             reserve(0)?;
121:         }
122:         Ok(self.deref_mut())
123:     }

138:             }
139:         }
140:     }
