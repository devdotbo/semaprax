impl<T: NominalElement> NominalPayload<T> {
    pub(super) fn detach_legacy_unfunded(&mut self) {
        let Storage::Shared(owner) = &mut self.storage else {
            unreachable!("legacy shared owner")
        };
        assert!(owner.mode == OwnerMode::LegacyUnfunded);
        let values = if let Some(owner) = Arc::get_mut(owner) {
            std::mem::take(&mut owner.values)
        } else {
            owner
                .values
                .iter()
                .map(NominalElement::clone_legacy_unfunded)
                .collect()
        };
        self.storage = Storage::Owned(values);
    }
    pub(crate) fn from_legacy_vec_unfunded(values: &mut Vec<T>) -> Self {
        let mut capacity = 0;
        for value in values.iter() {
            let required = value.drop_capacity(&mut |_| Ok::<_, std::convert::Infallible>(()));
            let required = match required {
                Ok(n) => n,
                Err(impossible) => match impossible {},
            };
            capacity = capacity.max(required);
        }
        // These are ordinary infallible legacy allocations, not a controlled
        // budget acquisition. In particular an existing active account pays 0.
        let scratch = Vec::with_capacity(capacity);
        Self {
            storage: Storage::Shared(Arc::new(Owner {
                mode: OwnerMode::LegacyUnfunded,
                values: std::mem::take(values),
                scratch,
            })),
        }
    }
}

//! Infallible legacy COW cloning, without recursive type-stack growth or payer.
use super::*;
fn ty(root: &ResolvedType) -> ResolvedType {
    enum Frame<'a> {
        Enter(&'a ResolvedType),
        Nominal(crate::hir::DeclarationId, usize),
        Function(usize),
    }
    let mut frames = vec![Frame::Enter(root)];
    let mut values = Vec::new();
    while let Some(frame) = frames.pop() {
        match frame {
            Frame::Enter(value) => match value {
                ResolvedType::Nominal {
                    declaration,
                    arguments,
                } => {
                    frames.push(Frame::Nominal(declaration.clone(), arguments.len()));
                    for child in arguments.iter().rev() {
                        frames.push(Frame::Enter(child));
                    }
                }
                ResolvedType::Function { parameters, result } => {
                    frames.push(Frame::Function(parameters.len()));
                    frames.push(Frame::Enter(result));
                    for child in parameters.iter().rev() {
                        frames.push(Frame::Enter(child));
                    }
                }
                leaf => values.push(leaf.clone()),
            },
            Frame::Nominal(declaration, count) => {
                let split = values
                    .len()
                    .checked_sub(count)
                    .expect("legacy nominal child count");
                let arguments = values.drain(split..).collect();
                values.push(ResolvedType::Nominal {
                    declaration,
                    arguments,
                });
            }
            Frame::Function(count) => {
                let result = Box::new(values.pop().expect("legacy function result"));
                let split = values
                    .len()
                    .checked_sub(count)
                    .expect("legacy function parameter count");
                let parameters = values.drain(split..).collect();
                values.push(ResolvedType::Function { parameters, result });
            }
        }
    }
    assert_eq!(values.len(), 1, "legacy complete cloned type");
    values.pop().unwrap()
}
