use std::{borrow::Cow, cmp::Ordering, mem::size_of, rc::Rc};
#[derive(Clone, Copy, Debug, Eq, PartialEq)]
struct Span {
    start: usize,
    end: usize,
    line: usize,
    column: usize,
}
#[derive(Clone, Copy, Debug, Eq, PartialEq)]
enum Severity {
    Error,
    Warning,
}
#[derive(Clone, Debug, Eq, PartialEq)]
struct Diagnostic {
    code: &'static str,
    severity: Severity,
    message: Cow<'static, str>,
    path: Option<String>,
    span: Option<Span>,
    help: Option<String>,
}
struct ResolvedFunction;
#[derive(Clone, Debug, Eq, PartialEq, Ord, PartialOrd)]
struct SkeletonObservation(u32);
#[derive(Clone, Debug, Eq, PartialEq, Ord, PartialOrd)]
enum SkeletonTerminal {
    Success,
    Failure,
}
struct Observations {
    prefix: Option<Rc<Vec<Rc<SkeletonObservation>>>>,
    items: Vec<Rc<SkeletonObservation>>,
}
struct SkeletonPath {
    observations: Observations,
    terminal: SkeletonTerminal,
}
struct ReplayBudget {
    remaining: usize,
    paid: usize,
    work: Vec<&'static str>,
    fees: Vec<(&'static str, usize)>,
}
impl ReplayBudget {
    fn charge(
        &mut self,
        f: &ResolvedFunction,
        n: usize,
        phase: &'static str,
    ) -> Result<(), Diagnostic> {
        self.work.push(phase);
        if self.remaining < n {
            return Err(replay_error(f, phase));
        }
        self.remaining -= n;
        Ok(())
    }
    fn reserve_bytes(
        &mut self,
        _f: &ResolvedFunction,
        n: usize,
        p: &'static str,
    ) -> Result<(), Diagnostic> {
        self.paid += n;
        self.fees.push((p, n));
        Ok(())
    }
    fn reserve_vec<T>(
        &mut self,
        f: &ResolvedFunction,
        v: &mut Vec<T>,
        n: usize,
        p: &'static str,
    ) -> Result<(), Diagnostic> {
        self.reserve_bytes(f, size_of::<T>() * n, p)?;
        v.reserve_exact(n);
        Ok(())
    }
}
fn replay_error(_f: &ResolvedFunction, message: &'static str) -> Diagnostic {
    Diagnostic {
        code: "SPX-H006",
        severity: Severity::Error,
        message: Cow::Borrowed(message),
        path: None,
        span: None,
        help: None,
    }
}
macro_rules! paid {
    ($b:expr,$f:expr,$n:expr,$p:literal $(,)?) => {{
        $b.reserve_bytes($f, $n, $p)
    }};
}
macro_rules! backing {
    ($b:expr,$f:expr,$v:expr,$n:expr,$p:literal $(,)?) => {{
        $b.reserve_vec($f, $v, $n, $p)
    }};
}

        pub struct Cursor<'a> {
            items: &'a [Rc<SkeletonObservation>],
            at: usize,
        }
        impl<'a> Cursor<'a> {
            pub fn new() -> Self {
                Self { items: &[], at: 0 }
            }
            pub fn prepare(
                &mut self,
                v: &'a Observations,
                _f: &ResolvedFunction,
                _b: &mut ReplayBudget,
            ) -> Result<(), Diagnostic> {
                assert!(v.prefix.is_none());
                self.items = &v.items;
                Ok(())
            }
            pub fn next(
                &mut self,
                _f: &ResolvedFunction,
                _b: &mut ReplayBudget,
            ) -> Result<Option<&'a SkeletonObservation>, Diagnostic> {
                let r = self.items.get(self.at);
                if r.is_some() {
                    self.at += 1;
                }
                Ok(r.map(Rc::as_ref))
            }
        }
    }

        impl Cursor {
            fn prepare(
                &mut self,
                history: &History,
                arena: &[Segment],
                function: &ResolvedFunction,
                budget: &mut ReplayBudget,
            ) -> Result<(), Diagnostic> {
                self.links.clear();
                self.segment = 0;
                self.item = 0;
                let mut previous = history.previous;
                while let Some(index) = previous {
                    budget.charge(function, 1, "ordinary plan history cursor traversal")?;
                    let segment = arena.get(index).ok_or_else(|| {
                        replay_error(function, "ordinary plan history link is invalid")
                    })?;
                    if self.links.len() == self.links.capacity()
                        || segment.previous.is_some_and(|parent| parent >= index)
                        || segment.depth
                            != self.links.first().map_or(segment.depth, |first| {
                                arena[*first].depth.saturating_sub(self.links.len())
                            })
                        || segment.depth == 0
                    {
                        return Err(replay_error(
                            function,
                            "ordinary plan history depth is invalid",
                        ));
                    }
                    self.links.push(index);
                    previous = segment.previous;
                }
                if let Some(first) = self.links.first() {
                    if arena[*first].depth != self.links.len() {
                        return Err(replay_error(
                            function,
                            "ordinary plan history depth is invalid",
                        ));
                    }
                }
                Ok(())
            }
            fn next<'a>(
                &mut self,
                history: &'a History,
                arena: &'a [Segment],
                function: &ResolvedFunction,
                budget: &mut ReplayBudget,
            ) -> Result<Option<&'a SkeletonObservation>, Diagnostic> {
                budget.charge(function, 1, "ordinary plan history forward cursor step")?;
                while self.segment < self.links.len() {
                    budget.charge(
                        function,
                        1,
                        "ordinary plan history forward segment traversal",
                    )?;
                    let items = &arena[self.links[self.links.len() - self.segment - 1]].items;
                    if let Some(value) = items.get(self.item) {
                        self.item += 1;
                        return Ok(Some(value));
                    }
                    self.segment += 1;
                    self.item = 0;
                }
                let value = history.tail.get(self.item);
                if value.is_some() {
                    self.item += 1;
                }
                Ok(value.map(|value| value.as_ref()))
            }
        }

        fn fixture(
            paths: usize,
            segments: usize,
            width: usize,
            tail: usize,
            mutation: usize,
        ) -> (HistorySet, Vec<SkeletonPath>) {
            let mut arena = Vec::new();
            let mut actual = Vec::new();
            let mut expected = Vec::new();
            let mut common: Option<(Option<usize>, Vec<Rc<SkeletonObservation>>)> = None;
            for path in (0..paths).rev() {
                let key = if mutation >= 3 { 0 } else { path };
                let mut previous = None;
                let mut all = Vec::new();
                if let Some((p, items)) = common.as_ref().filter(|_| mutation >= 5 && mutation != 9)
                {
                    previous = *p;
                    all = items.clone();
                } else {
                    for depth in 0..segments {
                        let segment_width = if (mutation == 7 && depth % 2 == 1)
                            || (mutation == 8 && depth % 2 == 0)
                            || (mutation == 9 && (path + depth) % 2 == 0)
                        {
                            0
                        } else {
                            width
                        };
                        let items: Vec<_> = (0..segment_width)
                            .map(|i| {
                                Rc::new(SkeletonObservation((key * 10000 + depth * 100 + i) as u32))
                            })
                            .collect();
                        all.extend(items.iter().cloned());
                        let at = arena.len();
                        arena.push(Segment {
                            previous,
                            items,
                            depth: depth + 1,
                        });
                        previous = Some(at);
                    }
                    if mutation >= 5 && mutation != 9 {
                        common = Some((previous, all.clone()));
                    }
                }
                let values: Vec<_> = (0..tail)
                    .map(|i| {
                        Rc::new(SkeletonObservation(
                            (key * 10000
                                + 5000
                                + i
                                + if (mutation == 4 || mutation == 6) && i + 1 == tail {
                                    path
                                } else {
                                    0
                                }) as u32,
                        ))
                    })
                    .collect();
                all.extend(values.iter().cloned());
                actual.push(PlanPath {
                    history: History {
                        previous,
                        tail: values,
                    },
                    terminal: SkeletonTerminal::Success,
                });
                expected.push(SkeletonPath {
                    observations: Observations {
                        prefix: None,
                        items: all,
                    },
                    terminal: SkeletonTerminal::Success,
                });
            }
            expected.sort_by(|a, b| a.observations.items.cmp(&b.observations.items));
            if paths > 0 && mutation == 1 {
                let last = expected.last_mut().unwrap();
                if last.observations.items.pop().is_none() {
                    last.observations
                        .items
                        .push(Rc::new(SkeletonObservation(42)));
                }
            }
            if paths > 0 && mutation == 2 {
                expected.last_mut().unwrap().terminal = SkeletonTerminal::Failure;
            }
            (
                HistorySet {
                    arena,
                    paths: actual,
                    max_depth: segments,
                },
                expected,
            )
        }
        fn observe(
            new: bool,
            paths: usize,
            segments: usize,
            width: usize,
            tail: usize,
            mutation: usize,
            cut: usize,
        ) -> (Result<(), Diagnostic>, ReplayBudget, usize) {
            let (actual, expected) = fixture(paths, segments, width, tail, mutation);
            let mut b = ReplayBudget {
                remaining: cut,
                paid: 0,
                work: Vec::new(),
                fees: Vec::new(),
            };
            let f = ResolvedFunction;
            let result = if new {
                after::compare(&f, &mut b, actual, &expected)
            } else {
                before::compare(&f, &mut b, actual, &expected)
            };
            let held = expected
                .iter()
                .flat_map(|p| p.observations.items.iter())
                .map(Rc::strong_count)
                .sum();
            (result, b, held)
        }

            let (new_owner, new_header, new_bank) = after::terms();
            assert_eq!(old_owner, new_owner);
            assert_eq!(header, new_header);
            assert_eq!((old_bank, new_bank), (40 * header, 42 * header));
            let setup = 2 * size_of::<&[Rc<SkeletonObservation>]>() + 3 * size_of::<usize>();
            let turn = 4 * size_of::<usize>() + size_of::<bool>();
            let finish = 4 * size_of::<&[Rc<SkeletonObservation>]>()
                + 4 * size_of::<usize>()
                + size_of::<Ordering>();
            assert_eq!((setup, turn, finish), (56, 33, 97));
            let flags = 3 * size_of::<bool>();
            let old_pattern = size_of::<&[Rc<SkeletonObservation>]>() + size_of::<bool>();
            let meta = 4 * size_of::<usize>() + 3 * size_of::<bool>();
            let old_advance = 3 * size_of::<&[Rc<SkeletonObservation>]>()
                + 2 * size_of::<usize>()
                + size_of::<Ordering>()
                + size_of::<bool>();
            let mut cases = 0;
            let mut cuts = 0;
            let mut runs_total = 0;
            let mut logical_steps = 0;
            for paths in [0, 1, 2, 3, 8] {
                for segments in [0, 1, 3, 16] {
                    for width in [0, 1, 2, 4, 16, 256] {
                        for tail in [0, 1, 16, 64, 256] {
                            for mutation in 0..10 {
                                let (a, old, oh) = observe(
                                    false,
                                    paths,
                                    segments,
                                    width,
                                    tail,
                                    mutation,
                                    usize::MAX,
                                );
                                let (b, new, nh) = observe(
                                    true,
                                    paths,
                                    segments,
                                    width,
                                    tail,
                                    mutation,
                                    usize::MAX,
                                );
                                assert_eq!(a, b);
                                assert_eq!(old.work, new.work);
                                assert_eq!(old.remaining, new.remaining);
                                assert_eq!(oh, nh);
                                let count = |budget: &ReplayBudget, amount| {
                                    budget
                                        .fees
                                        .iter()
                                        .filter(|(p, n)| {
                                            *p == "ordinary plan history paired cursor next"
                                                && *n == amount
                                        })
                                        .count()
                                };
                                let runs = count(&new, setup);
                                let turns = count(&new, turn);
                                let fast = count(&old, old_advance);
                                assert_eq!(count(&new, finish), runs);
                                assert_eq!(turns, fast + runs);
                                let empty = count(&old, size_of::<bool>());
                                let patterns =
                                    count(&new, size_of::<bool>()).checked_sub(empty).unwrap();
                                assert_eq!(count(&new, flags) + fast - runs, count(&old, flags));
                                assert_eq!(patterns + fast - runs, count(&old, old_pattern));
                                assert_eq!(count(&new, meta) + fast - runs, count(&old, meta));
                                for amount in [88, 98, 121, 124, 131, 154, 183] {
                                    assert_eq!(count(&old, amount), count(&new, amount));
                                }
                                let old_dynamic = flags * count(&old, flags)
                                    + old_pattern * count(&old, old_pattern)
                                    + meta * count(&old, meta)
                                    + old_advance * fast;
                                let new_dynamic = flags * count(&new, flags)
                                    + size_of::<bool>() * patterns
                                    + meta * count(&new, meta)
                                    + setup * runs
                                    + turn * turns
                                    + finish * runs;
                                let bank = if paths >= 2 { 2 * header } else { 0 };
                                assert_eq!(new.paid+old_dynamic,old.paid+bank+new_dynamic,"invoice paths={paths} segments={segments} width={width} tail={tail} mutation={mutation}");
                                runs_total += runs;
                                logical_steps += fast;
                                cases += 1;
                                if paths <= 3 && segments <= 3 && tail <= 16 && width <= 4 {
                                    for cut in 0..=old.work.len() {
                                        let (x, ob, oh) = observe(
                                            false, paths, segments, width, tail, mutation, cut,
                                        );
                                        let (y, nb, nh) = observe(
                                            true, paths, segments, width, tail, mutation, cut,
                                        );
                                        assert_eq!(x, y);
                                        assert_eq!(ob.work, nb.work);
                                        assert_eq!(ob.remaining, nb.remaining);
                                        assert_eq!(oh, nh);
                                        cuts += 1;
                                    }
                                }
                            }
                        }
                    }
                }
            }
            assert!(logical_steps > runs_total && runs_total > 0);
            println!("complete_current_run_cases={cases} work_cuts={cuts} shared_runs={runs_total} logical_shared_steps={logical_steps}; exact work/status/Rc and independent invoices; opaque model types; shared-HIR only typechecked; no full compiler/allocator/application fit claim");
        }
