module std.int.decimal;

// Exact unsigned decimal integers use canonical owned UTF-8 strings. Arithmetic
// is digit-wise; no complete operand is converted to a machine integer.
@id("std.int.decimal.digit")
fn digit(text: borrow str, index: i64) -> i64
    requires index >= 0 && index < str_len_bytes(text)
    ensures result >= -48 && result <= 207
{
    match str_byte_at(text, usize_from_i64(index)) { Option::Some { value: byte } => i64_from_u8(byte) - 48, Option::None {} => -48, }
}

// Return zero for an exhausted digit cursor; positive indices retain digit's contract.
@id("std.int.decimal.digit_or_zero")
fn digit_or_zero(text: borrow str, index: i64) -> i64
{
    if index < 0 { 0 } else { digit(text, index) }
}

// Nonempty ASCII digits, including leading zeros, are accepted for normalization.
@id("std.int.decimal.digits")
fn digits(text: borrow str) -> bool
{
    let n = str_len_bytes(text);
    let mut i = 0;
    let mut ok = n > 0;
    while i < n {
        let byte = digit(text, i);
        ok = digit_ok(ok, byte);
        i = i + 1;
        0
    }
    ok
}

// Canonical means exactly "0", or a nonzero digit followed by ASCII digits.
@id("std.int.decimal.valid")
fn valid(text: borrow str) -> bool
{
    digits(text) && canonical_head(text)
}

// Call digits first for untrusted input. Malformed input fails its precondition.
@id("std.int.decimal.canonicalize")
fn canonicalize(text: string) -> string
    requires digits(string_as_str(text))
    ensures string_len(result) > 0
{
    let n = string_len(text);
    let mut i = 0;
    while leading_zero(string_as_str(text), i, n) {
        i = i + 1;
        0
    }
    string_slice(text, i, n)
}

@id("std.int.decimal.compare")
fn compare(left: borrow str, right: borrow str) -> i64
    requires valid(left) && valid(right)
    ensures result >= -1 && result <= 1
{
    let n = str_len_bytes(left);
    let m = str_len_bytes(right);
    if n < m { -1 } else { if n > m { 1 } else { let mut i = 0; let mut out = 0; while comparison_pending(i, n, out) { let x = digit(left, i); let y = digit(right, i); out = x - y; i = i + 1; 0 } comparison_sign(out) } }
}

@id("std.int.decimal.add")
fn add(left: string, right: string) -> string
    requires valid(string_as_str(left)) && valid(string_as_str(right))
    ensures string_len(result) > 0
{
    let mut i = string_len(left) - 1;
    let mut j = string_len(right) - 1;
    let mut carry = 0;
    let mut out = "";
    while addition_pending(i, j, carry) {
        let x = digit_or_zero(string_as_str(left), i);
        let y = digit_or_zero(string_as_str(right), j);
        let sum = x + y + carry;
        out = string_concat(string_from_i64(sum % 10), out);
        carry = sum / 10;
        i = i - 1;
        j = j - 1;
        0
    }
    out
}

// Underflow is a checked contract failure. Compare before subtracting untrusted
// operands. The result is canonical even across a long borrow chain.
@id("std.int.decimal.subtract")
fn subtract(left: string, right: borrow str) -> string
    requires valid(string_as_str(left)) && valid(right)
    requires compare(string_as_str(left), right) >= 0
    ensures string_len(result) > 0
{
    let mut i = string_len(left) - 1;
    let mut j = str_len_bytes(right) - 1;
    let mut debt = 0;
    let mut out = "";
    while i >= 0 {
        let mut x = digit(string_as_str(left), i) - debt;
        let y = digit_or_zero(right, j);
        if x < y { x = x + 10; debt = 1; } else { debt = 0; }
        out = string_concat(string_from_i64(x - y), out);
        i = i - 1;
        j = j - 1;
        0
    }
    canonicalize(out)
}

// Long division returns the floor quotient. The partial remainder is shifted
// by one decimal place before adding each dividend digit. Since the previous
// remainder is below divisor, at most nine subtractions produce each digit.
@id("std.int.decimal.divide")
fn divide(dividend: string, divisor: borrow str) -> string
    requires valid(string_as_str(dividend)) && valid(divisor)
    requires digit(divisor, 0) != 0
    ensures string_len(result) > 0
{
    let n = string_len(dividend);
    let mut remainder = "0";
    let mut out = "";
    let mut i = 0;
    while i < n {
        let next = string_from_i64(digit(string_as_str(dividend), i));
        remainder = canonicalize(string_concat(remainder, next));
        let mut q = 0;
        while compare(string_as_str(remainder), divisor) >= 0 {
            remainder = subtract(remainder, divisor);
            q = q + 1;
            0
        }
        out = string_concat(out, string_from_i64(q));
        i = i + 1;
        0
    }
    canonicalize(out)
}

// Internal scalar decisions keep each caller's proof paths local to its loop.
@id("std.int.decimal.internal-digit-ok")
fn digit_ok(previous: bool, byte: i64) -> bool
{
    previous && byte >= 0 && byte <= 9
}

@id("std.int.decimal.internal-canonical-head")
fn canonical_head(text: borrow str) -> bool
{
    str_len_bytes(text) == 1 || digit(text, 0) != 0
}

@id("std.int.decimal.internal-leading-zero")
fn leading_zero(text: borrow str, index: i64, limit: i64) -> bool
{
    index < limit - 1 && digit(text, index) == 0
}

@id("std.int.decimal.internal-addition-pending")
fn addition_pending(left: i64, right: i64, carry: i64) -> bool
{
    left >= 0 || right >= 0 || carry > 0
}

@id("std.int.decimal.internal-comparison-pending")
fn comparison_pending(index: i64, limit: i64, out: i64) -> bool
{
    index < limit && out == 0
}

@id("std.int.decimal.internal-comparison-sign")
fn comparison_sign(value: i64) -> i64
{
    if value < 0 { -1 } else { if value > 0 { 1 } else { 0 } }
}
