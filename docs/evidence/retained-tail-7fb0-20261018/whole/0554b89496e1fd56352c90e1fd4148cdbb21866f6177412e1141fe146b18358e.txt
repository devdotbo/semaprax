//! Exact inverse of known source formatting; unknown byte changes still fail old checks.
//! The original archives and every original full-body comparison remain intact.
#[derive(Clone, Copy)]
pub(super) enum Kind {
    Root,
    OldCompare,
    BeforeLexical,
    BeforePair,
    Cursor,
    Tests880,
    Tests883,
}
pub(super) fn restore(kind: Kind, current: &str) -> String {
    let (length, prefix, edits) = match kind {
        Kind::Root => (18812, false, ROOT),
        Kind::OldCompare => (10345, false, OLDCOMPARE),
        Kind::BeforeLexical => (20151, false, BEFORELEXICAL),
        Kind::BeforePair => (21723, false, BEFOREPAIR),
        Kind::Cursor => (6323, false, CURSOR),
        Kind::Tests880 => (15444, true, TESTS880),
        Kind::Tests883 => (42934, true, TESTS883),
    };
    assert!(
        current.len() >= length && (prefix || current.len() == length),
        "unknown source extent"
    );
    let mut result = current.to_owned();
    let mut last = length;
    for &(at, after, before) in edits {
        let end = at
            .checked_add(after.len())
            .expect("source inverse overflow");
        assert!(end <= last && result.is_char_boundary(at) && result.is_char_boundary(end));
        assert_eq!(
            &result[at..end],
            after,
            "source drift in exact formatter inverse"
        );
        result.replace_range(at..end, before);
        last = at;
    }
    result
}

#[cfg(test)]
mod tests {
    use super::*;
    const CURRENT: &str = include_str!("../../../ordinary_plan_history.rs");
    const ORIGINAL: &str = include_str!(concat!(
        env!("CARGO_MANIFEST_DIR"),
        "/docs/evidence/opt883-lexical-plan-cursor-20261017/src__cleanup_plan__replay__ordinary_plan_history.rs.before.txt"
    ));

    #[test]
    fn inverse_preserves_unrecognized_body_drift_for_the_original_full_check() {
        assert_eq!(restore(Kind::Root, CURRENT), ORIGINAL);
        let changed = CURRENT.replacen("struct Cursor {", "struct Curxor {", 1);
        assert_ne!(changed, CURRENT);
        let restored = std::panic::catch_unwind(|| restore(Kind::Root, &changed));
        assert!(restored.is_err() || restored.unwrap() != ORIGINAL);
    }

    #[test]
    fn inverse_refuses_changed_known_hunks_and_unknown_source_extents() {
        let &(at, after, _) = ROOT.iter().find(|(_, after, _)| !after.is_empty()).unwrap();
        let mut changed = CURRENT.to_owned();
        let end = at + after.chars().next().unwrap().len_utf8();
        let replacement = if &changed[at..end] == "x" { "y" } else { "x" };
        changed.replace_range(at..end, replacement);
        assert!(std::panic::catch_unwind(|| restore(Kind::Root, &changed)).is_err());
        let longer = format!("{CURRENT}\n");
        assert!(std::panic::catch_unwind(|| restore(Kind::Root, &longer)).is_err());
    }
}
