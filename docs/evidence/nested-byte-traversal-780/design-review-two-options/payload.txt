fn frozen_body(
    body: &ResolvedExpr,
    payload: &ResolvedBinding,
    frozen: &[&ValueId],
) -> Result<(), Diagnostic> {
    walk(body, |expression, authorized| {
        if matches!(&expression.kind, ResolvedExprKind::Place(place) | ResolvedExprKind::BorrowPlace { place, .. } if place.root == payload.id)
        {
            return Err(fail("borrowed byte receiver payload escapes its receiver"));
        }
        if receiver_shape(expression).is_some() {
            return Err(fail(
                "borrowed byte traversal is nested in another traversal",
            ));
        }
        if let ResolvedExprKind::Block { statements, .. }
        | ResolvedExprKind::VecConstructor { statements, .. } = &expression.kind
        {
            for statement in statements {
                if let ResolvedStatement::Let { binding, .. }
                | ResolvedStatement::Assign { binding, .. } = statement
                {
                    if frozen.contains(&&binding.id) {
                        return Err(fail(
                            "borrowed byte traversal view, owner, counter or item is mutated or rebound",
                        ));
                    }
                }
            }
        }
        if let ResolvedExprKind::Match { arms, .. } = &expression.kind {
            for arm in arms {
                if let ResolvedMatchPattern::Variant { fields, .. } = &arm.pattern {
                    if fields
                        .iter()
                        .any(|field| frozen.contains(&&field.binding.id))
                    {
                        return Err(fail("borrowed byte traversal binding is shadowed"));
                    }
                }
            }
        }
        Ok(authorized)
    })?;
    Ok(())
}

pub(in crate::loan_plan) fn draft_bytes(
    expression: &ResolvedExpr,
    cfg: &Cfg<'_>,
    aliases: &impl crate::loan_plan::directory_access::Aliases,
    work: &mut WorkCounter,
) -> Result<Option<PaidDraft>, Diagnostic> {
    let Some(region) = crate::hir::byte_traversal_region(expression)? else {
        return Ok(None);
    };
    charge(work)?;
    let place = storage::expression_place(region.view, work.allocation)?
        .ok_or_else(|| error("borrowed byte traversal view lacks its exact place"))?;
    Ok(Some(PaidDraft {
        site: expression.id.clone(),
        origin: aliases.resolve(place, work)?,
        origin_span: region.view.span,
        parent_root: None,
        binding: None,
        start: cfg.node(expression, LoanPointPhase::Before)?,
        seeds: cfg_directory::singleton_set(
            cfg.node(expression, LoanPointPhase::After)?,
            work.allocation,
        )?,
        cause: LoanCause::TraversalFreeze,
    }))
}
