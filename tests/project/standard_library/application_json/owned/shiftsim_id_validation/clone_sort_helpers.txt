@id("shiftsim.schedule.unique_identifiers")
fn unique_identifiers(values: own Vec<string>) -> bool
{
    let sorted = vec_sort_owned<string>(values);
    let length = vec_len<string>(sorted);
    let mut index = 1usize;
    let mut valid = true;
    while valid && index < length {
        let previous = vec_str_at(sorted, index - 1usize);
        let current = vec_str_at(sorted, index);
        valid = previous != current;
        index = index + 1usize;
        valid && index < length
    }
    valid
}

@id("shiftsim.schedule.unique_patient_ids")
fn unique_patient_ids(values: borrow Vec<Patient>) -> bool
{
    let length = vec_len<Patient>(values);
    let mut identifiers = vec_with_capacity<string>(length);
    let mut index = 0usize;
    while index < length {
        let person_id_view = vec_field<Patient>(values, index, "id");
        identifiers = vec_push<string>(identifiers, string_from_str(person_id_view));
        index = index + 1usize;
        0
    }
    unique_identifiers(identifiers)
}
